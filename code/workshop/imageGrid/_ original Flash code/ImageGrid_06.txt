/*	***************
	CLASS MANAGER
	
	Controllers for adding and removing sets of cards; differentiated by the Class attribute 
	in the data.  Automaticaally loads the data from the server if it hasn't already been loaded.
		
		SetBtnState()			AddClass()
		ToggleClass()			OmitClass()
		BuildDrawStack()		UpdateVault()

		CleanStack()			Deprecated --> ClassMgr()
		
		ActiveClass()			Returns the index number of the Class in the ActiveClassesArray
	
	Classes are added to the Vault array by loading an xml file.  As of now, each xml data 
	file contains descriptions of one class of media ("Film", "Photo", etc.).  Each class
	is loaded into a temporary array (LoadStack), then transfered as an array into the Vault 
	along with a Class name and a Count of the number of records in that class.
	
	Note: In javascript, the Class might also refer to as Schema.  It's essentially the same thing,
	namely a group of related records or objects that share a common theme.  The Class (or Schema) 
	more or less divides content into different areas of interest.  These expressions are 
	not relevant to any programming terms.
	
	When a Class is "omitted", it just means existing cards of that type will 
	disappear from the stage.  The data records remain in the Vault array so cards can be re-added.
	OmitClass() calls for a Purge() of the DrawStack, the DiscardStack and the OnStage array.  For 
	the OnStage array, though, the cards being removed are only marked (Tag = true), not 
	removed from the array.  A marked card is deleted from the DiscardStack by ClassAssassin(), 
	which runs just before RecycleCards() moves the records back to the DrawStack.  This way, 
	you can re-add and re-omit cards immediately with no ill-effect.  (Theoretically)
		
	As of now, cards are not removed from the stage except by being bumped off by a new card,
	Omitting an entire Class of cards or "Brooming" the whole stage.  
		
		NOTE 1: There are descriptions of the Vault and Stack array structures at the bottom.  
		
		NOTE 2:  Classes match BtnNames associated with that Class.  Even so, buttons are 
		primarily identified by their ID number so the names can be generated at runtime and not 
		hand-coded.  The Button Name is an internal name.  The button legend is separate, 
		as is the button's instance name.  (UpdateBtn() accepts either an ID number
		or a Button Name.  The ID number is probably faster but haven't noticed a problem so 
		have resorted to BtnName, which results in cleaner code in most cases.)

/*	**********
	ADDCLASS()
	Adds a Class to the stage.  Automatically calls ImportXML() to load data into 
	the Vault first if needed, which will then add the contents to the DrawStack;
	Or, if the Class has already been loaded, skip that step and call 
	BuildDrawStack() directly.   */
	
	function AddClass (BtnIDNum:Number, TheClass:String) {
		var Load = true,
			i = 0;
		
		for (i in Vault){
		if (Vault[i][0] == TheClass){
			Load = false;
			}
		}
		
		if ( Load) {
			ImportXML (TheClass) ;
		} else {
			BuildDrawStack (TheClass) ;
		}
	}

/*	***********
	BUILDDRAWSTACK () 
	Transfers records from Vault to DrawStack; called by ImportXML() after loading Class data, 
	or directly by AddClass() if the the records are already in the Vault.  
	
	First, break the reference back to the Vault array.   */

	function BuildDrawStack (TheClass) {
	
			for (i in Vault ) {
				if (Vault[i][0] == TheClass) {
					n = Vault[i][2] ;
					
					for (r in Vault[i][1]) {
						
						var Cla:String	= Vault[i][1][r].Class,			//	}	Pick out the data; convert to strings.
							Cod:String  = Vault[i][1][r].Code,			//	}	We're breaking the reference back to the Vault
							Nam:String	= Vault[i][1][r].Name,			//	}
							Fil:String	= Vault[i][1][r].Filename,		//	}
							Lin:String	= Vault[i][1][r].Link;			//	}
						
						DrawStack.push ({ Class:Cla, Code:Cod, Name:Nam, Filename:Fil, Link:Lin, Tag:false }) ;
						//eek ("BuildDrawStack(): Added {" + Cla + ", " + Cod + ", " + Nam + ", " + Fil + ", " + Lin + "}" ) ;
					}
					
					ActiveClassesArray.push ([TheClass, n]) ;			//	Add the Class and the number of cards to the "Active" (on stage) array;
					UpdateClasses ("Active") ;							//	Re-calculates a string that contains the name(s) of the active class(es) <-- OBSOLETE
					CurrentClass = TheClass;							//	This is now the "current" class; GOTCHA: if not in RadioMode, a Class can be "active" without being the CurrentClass
					eek ("BuildDrawStack(): Added " + n + " " + TheClass + " Cards" ) ;
					break ;
				}
			}
			
			DealCards();
	}

/*	************
	OMITCLASS()
	Omits Cards of the specified Class; leaves records in the Vault un-touched;  
	!! --> Removed --> If AutoExit is OFF (false), test to make sure enough cards remain to fill the Grid.
	Farms out the dirty business to Purge().   */
	
	function OmitClass (TheClass) {	 

		eek ("OmitClass(): removing " + TheClass);
		
		for (i in ActiveClassesArray) {
			if (ActiveClassesArray[i][0] == TheClass) {
				ActiveClassesArray.splice(i,1) ;
				Purge (TheClass) ;
			}
		}
			
		UpdateClasses ("Active" ) ;
		UpdateBtn (HatBtnPad, TheClass, "OFF") ;
	}		


/*	**********
	UPDATECLASSES ()
	Updates VaultClasses or ActiveClasses; traces info about the classes in Vault; 
	updates from scratch on every call  */ 
	
	function UpdateClasses (Target:String) {
		
		var TextString:String = "" ;
		
		if (Target == "Vault") {
		
			VaultClasses = "" ;
			Comma = "" ;
			var len:Number = Vault.length ;
			eek ("The Vault contains:" ) ;
			
			for (i=0;  i < len; i++ ) {
				c = Vault[i][0] ;
				n = Vault[i][2] ;
				eek ( "\t" + n + " \t" + c + "Cards ") ;
				if (i > 0) { Comma = "," ; }
				VaultClasses = VaultClasses + Comma + Vault[i][0] ;
			}
			
		} else if (Target == "Active") {								//	<-- Obsolete; only useful for dev panel
			
			ActiveClasses = ActiveClassesArray.toString() ;				//	<-- Obsolete; use ActiveClass(theClass) instead
			eek ("UpdateClasses(): Active classes: " + ActiveClassesArray.length ) ;
		
		} else {
			
			eek ("  !! UpdateClasses(): unknown parameter  !!  " ) ;
		}
	}	
	
	
/*	************

	var ActiveClass = function(theClass) {
		var i;

			}			
		}

/*	***********
	PURGE()  
	Remove records of the specified Class from the specified Stack;
	note special rules for cards that are on stage  */

	function Purge (TheClass:String) {
		TagClass (DrawStack, TheClass, "Draw Stack") ;	
		TagClass (OnStage, TheClass, "OnStage Stack") ;
		TagClass (DiscardStack, TheClass, "Discard Stack") ;
		CleanStack (DrawStack, TheClass, "Draw Stack") ;
		CleanStack (DiscardStack, TheClass, "Discard Stack") ; 
	}
	
	
/*	*************
	TAGCLASS()
	Tag cards of a Class in a CardStack (usually to be ommitted).  */ 
	
	function TagClass (TheStack, TheClass:String, TheStackName:String) {
		n = 0 ;
		for (i in TheStack ) {
			if (TheStack[i].Class == TheClass) {
				TheStack[i].Tag = true ;
				n++ ;
			}
		}
		eek ( "	Tagged  " + n + " " + TheClass + "Cards for removal from the " + TheStackName) ;
	}
			
			
/*	*************
	CLEANSTACK()
	Remove cards that have been tagged OR are of a certain Class.  Seems like overkill
	but cards keep slipping through.  Don't use this on the OnStage array.  Use ClassAssassin() instead. */
	
	function CleanStack (TheStack, TheClass, TheStackName:String) {
		n = 0 ;
		for ( i in TheStack ) {
			len = TheStack.length ;
			for (r=0; r< len ; r++) {
				if (TheStack[r].Class == TheClass || TheStack[r].Tag == true) {
					TheStack.splice(r,1) ;					//	delete the card from the stack
					n++
				}
			}
		}
		eek ("	Omitted " + n + " " + TheClass + "Cards from the " + TheStackName ) ;   
	}

/*	*************
	SWEEPSTAGE()
	Yet another hack to zap stray cards left after a Flush.  Pesky buggers.  
	This one is looking for orphaned clip instances, maybe stranded during a FlushCards()
	operation.  Still not 100% success... */
	
	function SweepStage(callback) {
		
		c = CellCount * 5 ;						//	include cards orphaned during MakeCard (with higher depth than normal)
		
		for (i=0; i<c; i++) {
			n = undefined ;
			n = this.getInstanceAtDepth (i) ;				//	identify any strays at this depth
			if ( n ) {
				n.removeMovieClip() ;						//	remove it
				eek ("  !!  Removed stray card in cell " + i + "  !!" ) ;
			}
		}

		if (callback){
			callback();
		}
	}
