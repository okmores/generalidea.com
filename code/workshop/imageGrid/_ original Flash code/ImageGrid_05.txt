/*	*******************
	THROTTLE()
	Controls the speed of dealing cards, including "off" (or actually, "manual"). */
	
	function Throttle (Input:String) {
		
		l = false ;								//	Should we be in looping mode (AutoLoad)?
		t = false ;								//	Trigger a New Card?
		f = false ;								//	Flash?
		s = false ;								//	update BufferSave for a restore?
		r = false ;								//	Restore the saved buffer, trigger a BtnHit (if it's needed)
		
		ID = undefined ;
		
		switch (Input) {
			case "Pause" :						//	Remember: options above remain false unless explicitly changed		
				Paused = true ;					//	MC() will exit on finding this 
				AutoLoad = false ;
				break ;
				
			case "UnPause" :
				Paused = false ;
				r = true ;						//	Get the saved buffer, trigger a card
				break ;
				
			case "Off" :
				Buffer = -1 ;
				s = true ;
				break ;
				
			case "Manual" :
				Buffer = -1 ;
				t = true ;
				s = true ;
				break ;
				
			case "Slow" :
				Buffer = BufferMax ;
				l = true ;
				t = true ;
				f = true ;
				s = true ;
				break ;
				
			case "Medium" :
				Buffer = BufferMax/2 ;
				l = true ;
				t = true ;
				f = true ;
				s = true ;
				break ;
				
			case "Fast" :
				Buffer = BufferMax/4 ;
				l = true ;
				t = true ;
				f = true ;
				s = true ;
				break ;
				
			case "Fastest" :
				Buffer = BufferMax/10 ;
				l = true ;
				t = true ;
				s = true ;
				break ;
				
			case "Flash" :
				Flash () ;						//	Flash the entire Grid (the default)
				l = null ;						//	don't change the looping mode (we're ovrriding it )				
				t = true ;						//	run even if we're stopped
				break ;	

			default :
				eek ("  !!  Throttle() - unknown parameter " ) ;
		}
		
		if (Input != "Flash") { BufferSpeed = Input ; }
		
		if ( s ) { BufferSave = Input ; }

		if ( l == null ) {
			break ;
		} else if ( l == true ) {
			AutoState = "Multiple" ;
		} else {
			AutoState = "Single" ;
		}
		
		if ( f ) { 
			if (!FlashMode) {					//	If not already in a Flash situation  (ie, don't reset an ongoing Flash)
				Flash (1) ;
			}
		}
		
		if ( t ) { 
			MC("NewCard") ;
		}
		
		if ( r ) {
			//Throttle (BufferSave) ;
			//UpdateBtnGroup ( "ShowBtns", BufferSave ) ;
			ID = Number ( ShowBtnPad.GetBtnID (BufferSave) ) ;
			BtnHit (ID,BufferSave) ;		
		}
	}
	
	
/*	*************
	RUNSHOW() 
	Set up the stage to run the show from scratch */
	
	function RunShow() {
		HeaderPoof = true ;
		ShowTime = true ;
		BtnHit (BufferSave,BufferSave) ;		//	(BtnID, BtnName); retrieve and use data from previously saved state
		Flash() ;								//	Flash load until the Grid is full
	}

	BROOM()
	Remove all cards from the stage;
	used for calls originating outside of Flash  */
	
	function Broom(onFinish){
		if (ShowTime){
			FlushCards ("Broom") ;
		} else {
			
		}
	}


/*	************
	FLUSHCARDS()
	Make all the cards go away ; basically a Reset, but prettier;
	onFinish contains instructions to accomplish when done:
	onFinish == "Broom" indicates a simple end of the show;
	onFinish ==  anything else -- assume it's a link to an asset  */
	
	function FlushCards (onFinish) {
		
		AppendTasks.push (["FlushCards", onFinish ]) ;			//	do onFinish when done
		
		eek ("FlushCards")
		Reset = true ;
		clearTimeout(BufferTime) ;
		ShowTime = false ;
		AutoLoad = false ;
		CurrentClass = "";
		ActiveClassesArray.length = 0 ;
		ActiveClasses = "" ;
		DrawStack.length = 0 ;
		DiscardStack.length = 0 ;
		OnStage.length = 0 ;
		UpdateBtnGroup (ShowBtnPad, "ShowBtns", undefined, "INVISIBLE") ;  //	(BtnPad, BtnGroup, OnBtn [, ForceReset])
		ActiveCards = 0 ;
		//Quorum = false ;
		SetCells () ;
		PoofCard () ;
	}
	
	function PoofCard () {									//	PoofCard is a slave of FlushCards 
															//	This will loop until all the cards are removed
		var len:Number = Grid.length ;						//	Removing array elements so must recalc every iteration

		if (len > 0) {										//	check to see if all cards are gone
			i = random (len) ;								//	pick a random card to erase
			n = Grid[i] ;									//	
			Card = this.getInstanceAtDepth (n) ;			//	Find the card in this cell number
			Card.removeMovieClip() ;						//	remove it
			Grid.splice(i,1) ;								//	remove the record in the array
			var PoofTimer:Number = setTimeout (PoofCardBuffer, PoofBuffer ) ;	// slow down; set speed in frame 1 vars
		
		} else {											//	Done flushing cards...
			SweepStage() ;									//	Check for stray cards, the little creeps
			Append ("FlushCards") ; 						//	Check to see if there's anything else to do
			Reset = false ;									//	update the state
			eek ("==== Stage cleared ====" ) ;

		}		
	}

	function PoofCardBuffer () {
		PoofCard()
	}
	
	
/*	*************
	Append()
	Arbitrarily add additional processing steps to a function; A clunky mechanism for a future 
	callback along the lines of a Deferred promise in jQuery.
	Functions must provide a way to look for an appended task before finishing.  */
	
	var AppendTasks:Array = new Array() ;
	
	function Append (TheCaller:String) {
		
		var task = undefined ;
		
		for ( i in AppendTasks ) {								//	check to see if there's anything for the caller
			
			if (AppendTasks[i][0] == TheCaller ) {
				switch (TheCaller) {
					case "FlushCards" :
						
						task = AppendTasks[i][1] ;						
						
						if (task == "Broom") {					//	"BROOM" NO MORE ACTIONS
							if (gotJavascript){					//	but need to tell javascript about it
								js ("FlashBroom");
							}
							eek ("  Broom complete" );
						
						} else {								//	"LINK" 
																//	the task is an asset name or code;
							eek ("  Broom complete, asking for asset..." );    
							if (Host == "Javascript") {			//	if we're in a javascript environment...
								js ("Get", task); 				//	ask javascript to go get it
							} else if (Host == "Flash") {  		//	otherwise
								Player.GetMovie (task) ;		//	ask the attached Flash Player to play it
							} else {
								eek ("Append(): ERROR - unknown Host" )
							}
						}
						
						AppendTasks.splice (i,1) ;				//	remove this task from the array
						break ;

					default :
						eek ("Append():  !!  unknown caller  !!" ) ;
				}
			} else {
				eek ("Append(): No tasks for " + TheCaller ) ; 
			}
		}
	}			


/*	**********
	FLASH() & REFLASH()
	Flash() fills the Grid as fast as possible with CellCount number of cards.  Reflash() is located at the end of MC() 
	and checks to see if it's time to end the Flashing state.  Flash() is also used to "flash" one or more 
	cards to make the program feel more responsive (overrides the buffer on a button press).  A Flash state 
	does not change the Buffer settings, it overrides the Throttle and whatever settings are present,  
	including settings associated with the OFF state.  */
	
	function Flash(NumCards:Number) {
		if (!NumCards) { SetCells() ; }									//	if a full Flash, make all cells available
		UpdateBtn (ShowBtnPad, "Flash", "ON") ;							//	because Flash() overrides the Buffer anyway, we're only changing the Btn appearance
		FlashMode = true ;
		FlashCards = NumCards || CellCount ;
	}
	
	function ReFlash() {
		if (FlashCards == 0 ) {
			FlashMode = false ;	
			UpdateBtnGroup (ShowBtnPad, "ShowBtns", BufferSave ) ;		//	because Flash() overrides the Buffer, we're only changing the Btn appearance
		}
	}

/*	**********
	POOF() & UNPOOF()
	Make this whole mess disappear or reappear  */
	
	function UnPoof () {
		eek ("UnPoof()");
		js ("eek", "ImageGrid.UnPoof()");
		this._visible = true;
		if (ShowHatBtns) { this.HatBtnPad._visible = true; }
		this.ShowBtnPad._visible = true;
	}
	
	function Poof () {
		js ("eek", "ImageGrid.Poof()");
		this._visible = false ;
		this.HatBtnPad._visible = false;
		this.ShowBtnPad._visible = false;
	}


/*	*************
	SETRADIOMODE()
	RadioMode(false) allows more that one class to be on stage at a time.
	This just sets a flag in vars. */
	
	function SetRadioMode(newMode){
		if (newMode == "Toggle"){
			if (RadioMode){
				RadioMode = false;
			}else{
				RadioMode = true;
			}
		}else{
			RadioMode = newMode;
		}
	} 


/*	************
	HOTSWITCH()
	Load the requested Class of cards immediately, using the ImageGrid Flash() 
	method; HotSwitch() replaces any other Class(es) on stage, so this should only be 
	used in RadioMode.  */

			for (i in ActiveClassesArray){

				}
