/*	**********
	IMPORTXML()
	Imports XML data source file; you must set up any appropriate vars before calling this  
	
		THE XML FILE MUST USE THE WORD "Card" FOR EACH NODE
		Apparently, the word is used to create a unique name for each protocard
		something like "Card24" or something.  I don't have time to investigate this
		right now but look at line 50 below.  Sheese.  */
	
	function ImportXML (TheClass:String) {
	
		ClassName = TheClass ;
		eek ("ImportXML(): " + ClassName);
		if (TheClass == "Clients"){ClassName = "Logo";}			//	!!  HACK  !!  HTML Class for "Logo" might be "Clients"
		if (TheClass == "Ads"){ClassName = "Ad";}				//	!!  HACK  !!  Accept "Ads" or "Ad"
		
		var p:String = RootPath_XML ;
		var f:String = ClassName + "Cards.xml" ;				//	using "ClassName" because "Class" is a protected word in ActionScript 
		
		//if (TheClass == "Film"){f = "FilmCards_.xml";}		//	*****!!!!!!!!!!!! DEFEAT BROWSER CACHE !!!!!!!!!!!!!

		var FullPath:String = p + f ;
		XMLParser.load (FullPath, onImportXML);
		eek ("ImportXML(): " + FullPath);
		js ("eek", "ImportXML(): " + FullPath);
	}
	
	//	Listener
	
	function onImportXML ($success:Boolean, $parsedObject:Object, $xml:XML) { 
		
		if ($success) {
			
			LoadStack = $parsedObject.Card ;					//	"Card" is the name of the data record (each of them) in the xml document; ie, every data record is a "Card", which has various attributes like "Name", Code", etc.
			//ListAllItems (LoadStack, "items loaded") ;		//	Clunky, but arrays don't have names -- param 2 is only used to trace
			Vault.push ( [ClassName, LoadStack, LoadStack.length] ) ;
			BuildDrawStack (ClassName) ;
			UpdateClasses ("Vault") ;
		
		} else {
			
			eek ("ImportXML():  !!   failed  !!") ;
			js ("eek", "ImportXML(): !!   failed  !!");
		}
	}

		}

/*	**********
	LISTVAULT () */
	
	function ListVault() {
		for (i in Vault) {
			c = Vault[i][1][0].Class ;
			t = c + " records in Vault" ;
			a = Vault[i][1] ;
			ListAllItems (a, t) ;
		}
	}

	After an array element is used, that array member is deleted.  This way, only unused cell numbers
