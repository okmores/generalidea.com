/*	**************

	be defeated, though, since the "Caller" parameter expects an internal btn name, not legend text.

					string not a method or object -- see above), so we have to supply the instance name

		var i = BtnIDNum,

				if (gotJavascript){
					FlushCards (link) ;						//	FlushCards() will check for the link when it gets done

					eek ("  BtnHit() !! EXIT !!  No Javascript; FLV Video is off; SWF Player does not recognise media by CODE instead of FILENAME" ) ;
				}
							
				// PAUSE the cards and leave them on the Stage

					// Do something (like darken the cards)
				
				break;
				
			case "Ads" :								//	} !!  HACK  !! Fix to match javascript terms

			case "Clients" :							//	} !!  HACK  !! Beware "Clients" and "Logo" confusion; means the same thing. (may have been fixed)

			case "Ad" :

			case "Photo" :								//	} need to automate;
			case "Logo" :								//	} Beware "Clients" and "Logo" confusion; means the same thing. (may have been fixed)
				
				if (ShowTime) { 						//	IF A SHOW IS RUNNING:
					if (RadioMode) {					//		RADIO BTNS ON (Only 1 Class on stage at a time)

/*	***************
	UPDATEBTN()
	Update a btn; see notes above;
	Accepts btn name as string or ID number */

	function UpdateBtn (TheBtnPad, TheBtn, TheState:String) {
		
		var BtnPadName = TheBtnPad._name ;
		var ID:Number = null ;
		var bName:String = null ;
		
		if ( typeof (TheBtn) == "string" ) {
			ID = TheBtnPad.GetBtnID (TheBtn) ;
			bName = TheBtn ;							//	for Javascript external, which knows not of the internal ID number
			//eek ("UpdateBtn():" + BtnPadName + ".UpdateBtn() - ID supplied was the string " + TheBtn + "; converted to " + ID ) ;
		} else if ( typeof (TheBtn) == "number" ) { 
			ID = TheBtn ;
			//eek ("UpdateBtn():" + BtnPadName + ".UpdateBtn() - ID supplied was the number " + ID ) ;
		} else {
			eek ("UpdateBtn():" + BtnPadName + ".UpdateBtn() - ID supplied \"" + TheBtn + "\" resolves to ID number: " + ID ) ;
		}
		
		TheBtnPad.SetState ( ID, TheState ) ;
		
		// synchronize btns in html
		// no javascript written to handle this?  It doesn't work but maybe it's not needed now since all controls are in html.
		if (gotJavascript && BtnPadName == "HatBtnPad") {
			p1 = BtnPadName ;
			p2 = bName || TheBtnPad.GetBtnName(ID) ;
			p3 = TheState ;
			js ("eek", "UpdateBtn(): " + p1 + " " + p2 + " " + p3);
			js (p1,p2,p3) ;								//	beware -- no error trap 
		}
	}
	
	
/*	***************
	UPDATEBTNGOUP()
	Review all btns in a RADIO BtnGroup. Optionally turn one of them ON.  
	NOTE --> Use also to switch btns to LOCKED or INVISIBLE.  (See PoofBtnPad() and UnPoofBtnPad().  */

	function UpdateBtnGroup (TheBtnPad, BtnGroupName:String, TheOnBtn, NewState:String) {		//	
		
		var BtnPadName = TheBtnPad._name ;
		var n:String = BtnGroupName ;
		var TheState:String = "OFF" ;
		var OnID:Number = undefined ;
		var BtnIDNum:Number = undefined ;
		
		if (NewState == "LOCKED" || NewState == "INVISIBLE" || NewState == "HIDDEN" ) {	
			TheState = NewState ;
		}
		
		if (TheOnBtn == "Current") {TheOnBtn = CurrentClass;} 
		
		var RadioBtns = [];								//	!!  HACK  !!  hand set !!
		if (BtnGroupName == "HatBtns") {				//	!!  should read the PushBtns array instead but  !!
			RadioBtns = HatBtns;						//	!!	it ain't workin  !!
		}else{
			RadioBtns = ShowBtns;
		}		

		if ( typeof (TheOnBtn) == "string" ) {
			OnID = TheBtnPad.GetBtnID (TheOnBtn) ;
		} else if ( typeof (TheOnBtn) == "number" ) {
			OnID = TheOnBtn ;
		} else if ( TheOnBtn == undefined ) {			//	?  PushBtns = HatBtns,0,1,6,HatBtnPad,ShowBtns,0,1,2,3,4,5,6,ShowBtnPad
			OnID = null ;
		} else {
			OnID = null ;
		}
		
		for (i in RadioBtns) {
			j = RadioBtns[i] ;
			UpdateBtn (TheBtnPad, j, TheState ) ;		//	eek ("Update " + TheBtnPad + "[" + j + "] " + TheState + " <--");
		}
		
		m = "Updated " + BtnPadName + "." + n ;
		
		if (OnID != null) {
			UpdateBtn (TheBtnPad, Number(OnID), "ON") ;
			m += ";  Btn-" + OnID + " selected" ;
		}
		
		eek (m) ;
		//TheBtnPad.ListBtns() ;
	}
